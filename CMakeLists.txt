cmake_minimum_required(VERSION 3.20)
project(SimpleVideoHandle VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "SimpleVideoHandle requires Windows x64")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Configure an x64 compiler (Visual Studio: -A x64)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(video_core STATIC src/engine.cpp)
target_include_directories(video_core PUBLIC src)
target_compile_definitions(video_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(video_core PUBLIC ole32)
add_executable(SimpleVideoHandle WIN32 src/main.cpp src/app.rc)
target_link_libraries(SimpleVideoHandle PRIVATE video_core comctl32 shell32 comdlg32 uuid gdi32 user32)
add_executable(SimpleVideoHandleCLI src/cli.cpp)
target_link_libraries(SimpleVideoHandleCLI PRIVATE video_core)
if(MSVC)
  foreach(target video_core SimpleVideoHandle SimpleVideoHandleCLI)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive- /EHsc $<$<CONFIG:Release>:/O2>)
  endforeach()
  # Our complete manifest is embedded by the resource compiler.
  target_link_options(SimpleVideoHandle PRIVATE /MANIFEST:NO)
elseif(MINGW)
  foreach(target SimpleVideoHandle SimpleVideoHandleCLI)
    target_link_options(${target} PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endforeach()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE video_core shell32)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8 /W4 /permissive- /EHsc)
  elseif(MINGW)
    target_link_options(core_tests PRIVATE -municode -static -static-libgcc -static-libstdc++)
  endif()
  add_test(NAME core_tests COMMAND core_tests)
endif()
install(TARGETS SimpleVideoHandle SimpleVideoHandleCLI RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
